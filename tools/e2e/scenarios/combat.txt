# Сценарий: боевые замеры (DPS + лечение) и настраиваемое название команды.
#
# Замеры запускает не чат, а флаг-файл run/aoe-devtest.flag (его создаёт шаг
# «Очистить прошлые снимки» в workflow): как только игрок оказывается в мире,
# CombatProbe сам наносит 10 ударов по манекену через DamageEvent, лечит игрока
# через ResourcesData.restore(health), проверяет восстановление маны и пишет в
# лог строки AOETEST. Читы в мире для этого не нужны.
#
# Название команды зонд тоже проверяет сам (TeamData.createTeam + чистка имени),
# а HUD показывает его в панели «AOE / ОБЗОР» — это видно на снимке 05.
#
# Координаты — под окно 1280x720 (run/options.txt) и guiScale 2.

wait_window Minecraft 420000
wait_log 300000 Created: .*atlas/items
wait 8000
shot 00-main-menu

# Главное меню → Одиночная игра → список миров
move 640 296
click 640 296
wait 4000
shot 01-world-list

# «Создать новый мир» (внизу слева) → экран создания
move 480 688
click 480 688
wait 3000
shot 02-create-world

# Создаём мир
move 480 688
click 480 688
wait 45000
shot 03-in-world

# --- замеры DPS и лечения ---------------------------------------------
# Зонд стартует сам по флагу; полный прогон — доли секунды плюс запись итога.
wait_log 120000 AOETEST summary
wait 400
# Бой ещё «горит» (окно 6 с) — панель должна показывать DPS.
shot 04-combat-hud
# Ждём больше FIGHT_TIMEOUT_MS, панель должна вернуться к обзору с названием
# команды «Тихий авангард» (его выставил зонд).
wait 8000
shot 05-after-fight

# --- настройки HUD: переключение видимости горячей клавишей -------------
hold o 400
wait 1500
shot 06-hud-hidden
hold o 400
wait 1500
shot 07-hud-shown

# Смена позиции панели
hold p 400
wait 1500
shot 08-hud-position

jstack combat
log конец сценария combat
