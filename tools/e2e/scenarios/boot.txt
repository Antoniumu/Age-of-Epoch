# Сценарий загрузки dev-клиента.
# Цель первого прогона: доказать, что клиент стартует и рисует главное меню,
# и снять заведомо безопасные шаги. Дальше по снимкам подбираются координаты.
#
# Формат команд описан в tools/e2e/run_scenario.py.

wait_window Minecraft 420000
# Маркер готовности: звук на раннере отключён (null-драйвер), а строка
# "Sound engine started" зависит от аудиоустройства. Поэтому ждём атласы —
# они грузятся уже после создания окна и GPU-контекста.
wait_log 300000 Created: .*atlas/items
windows
wmctrl
wait 8000
shot 00-main-menu
jstack after-first-shot

# Главное меню: «Одиночная игра» — первая кнопка сверху.
# Координаты считаются от размера окна из run/options.txt (1280x720, guiScale 2).
click 640 296
wait 3000
shot 01-world-list

log конец сценария загрузки
