# Сценарий: боевые замеры (DPS + лечение) и настраиваемое название команды.
#
# Замеры запускает не чат, а файл-триггер run/aoe-devtest.flag: CombatProbe
# видит его, наносит 10 ударов по манекену через DamageEvent, лечит игрока через
# ResourcesData.restore(health), проверяет cap маны и пишет в лог строки AOETEST.
# Команда `touch` ниже пересоздаёт флаг в нужный момент, поэтому снимок попадает
# ровно в бой. Читы в мире для этого не нужны.
#
# Название команды зонд тоже проверяет сам (TeamData.createTeam + чистка имени),
# а HUD показывает его в панели «AOE / ОБЗОР» — это видно на снимке 05.
#
# Координаты — под окно 1280x720 (run/options.txt) и guiScale 2.

wait_window Minecraft 420000
wait_log 300000 Created: .*atlas/items
wait 8000
shot 00-main-menu

# Главное меню → Одиночная игра → список миров
move 640 296
click 640 296
wait 4000
shot 01-world-list

# «Создать новый мир» (внизу слева) → экран создания
move 480 688
click 480 688
wait 3000
shot 02-create-world

# Создаём мир
move 480 688
click 480 688
wait 45000
shot 03-in-world

# --- замеры DPS и лечения ---------------------------------------------
# Первый прогон зонд делает сам при входе в мир (флаг ставит workflow), но это
# происходит до снимков. Поэтому перезапускаем его, когда всё готово: полный
# прогон занимает доли секунды, бой «горит» ещё 6 с.
touch run/aoe-devtest.flag
wait 900
# Бой активен — панель должна показывать «AOE / БОЙ» с DPS.
shot 04-combat-hud
wait 1500
wait_log 60000 AOETEST summary
wait 300
# Ждём больше FIGHT_TIMEOUT_MS, панель должна вернуться к обзору с названием
# команды «Тихий авангард» (его выставил зонд).
wait 8000
shot 05-after-fight

# --- бесдушные основы: предметы из зонда лежат в хотбаре -----------------
# Зонд SoullessProbe кладёт в слоты 1-4: бесдушный деревянный меч, бесдушный
# незеритовый меч, тот же деревянный с наложенной душой и саму душу. Открываем
# инвентарь и наводим курсор на слоты — в тултипе видно «Бесдушная основа»
# и разницу статов между дешёвой и дорогой основой.
key E
wait 1500
move 620 480
wait 1200
shot 09-inventory-soulless
move 660 480
wait 1200
shot 10-inventory-rich-base
move 700 480
wait 1200
shot 11-inventory-souled
key E
wait 800

# --- настройки HUD: горячие клавиши F7 (скрыть) и F8 (положение) ---------
hold F7 400
wait 1500
shot 12-hud-hidden
hold F7 400
wait 1500
shot 13-hud-shown

# Смена позиции панели
hold F8 400
wait 1500
shot 14-hud-position

jstack combat
log конец сценария combat
