# Сценарий: боевые замеры (DPS + лечение), экипировка и макс. HP, название команды.
#
# Замеры запускает не чат, а файл-триггер run/aoe-devtest.flag: CombatProbe
# видит его, наносит 10 ударов по манекену через DamageEvent, лечит игрока через
# ResourcesData.restore(health), проверяет cap маны, надевает броню со статом
# здоровья и пишет в лог строки AOETEST.
# Команда `touch` ниже пересоздаёт флаг в нужный момент, поэтому снимок попадает
# ровно в бой. Читы в мире для этого не нужны.
#
# Стадия gear — регрессия на «броня со статами не поднимает макс. HP до
# перезахода»: зонд надевает нагрудник с аффиксом здоровья, ждёт пару тиков
# (штатный пересчёт, без forceRecalculateStats), сверяет макс. HP и ~8 секунд
# держит броню надетой — HUD с выросшим «Здоровье: X/Y» попадает в кадр 05.
#
# Название команды зонд тоже проверяет сам (TeamData.createTeam + чистка имени),
# а HUD показывает его в панели «AOE / ОБЗОР» — это видно на снимке 07.
#
# Координаты — под окно 1280x720 (run/options.txt) и guiScale 2.

wait_window Minecraft 420000
wait_log 300000 Created: .*atlas/items
wait 8000
shot 00-main-menu

# Главное меню → Одиночная игра → список миров
move 640 296
click 640 296
wait 4000
shot 01-world-list

# «Создать новый мир» (внизу слева) → экран создания
move 480 688
click 480 688
wait 3000
shot 02-create-world

# Создаём мир
move 480 688
click 480 688
wait 45000
shot 03-in-world

# --- замеры DPS и лечения ---------------------------------------------
# Первый прогон зонд делает сам при входе в мир (флаг ставит workflow), но это
# происходит до снимков. Поэтому перезапускаем его, когда всё готово.
# log_mark ставим ДО триггера: wait_log_new ждёт строки именно этого прогона
# (обычный wait_log нашёл бы строки первого и вернулся сразу).
# Порядок стадий: dps (~1 с) → heal → team → gear (надели, держим ~8 с,
# сняли, прокрутили на себе) → summary.
log_mark
touch run/aoe-devtest.flag
# Бой закончен — панель должна показывать «AOE / БОЙ» с DPS (окно 6 с после
# последнего удара).
wait_log_new 60000 AOETEST dps_avg_hit
wait 200
shot 04-combat-hud

# --- экипировка и макс. HP ---------------------------------------------
# Регрессия на «броня со статами не поднимает макс. HP до перезахода».
# Кадр 05 снимаем сразу после строки gear_equip: зонд держит броню надетой
# ещё ~8 с игрового времени, кадр гарантированно попадает в это окно.
# Максимум здоровья в HUD «Здоровье: X/Y» должен быть выше, чем без брони.
wait_log_new 60000 AOETEST gear_equip
wait 200
shot 05-gear-equip-max-hp
# Кадр 06 — после gear_unequip максимум вернулся к исходному.
wait_log_new 60000 AOETEST gear_unequip
wait 200
shot 06-gear-unequip-max-hp

wait_log_new 60000 AOETEST summary
wait 300
# Ждём больше FIGHT_TIMEOUT_MS, панель должна вернуться к обзору с названием
# команды «Тихий авангард» (его выставил зонд).
wait 8000
shot 07-after-fight

# --- настройки HUD: горячие клавиши F7 (скрыть) и F8 (положение) ---------
hold F7 400
wait 1500
shot 08-hud-hidden
hold F7 400
wait 1500
shot 09-hud-shown

# Смена позиции панели
hold F8 400
wait 1500
shot 10-hud-position

jstack combat
log конец сценария combat
