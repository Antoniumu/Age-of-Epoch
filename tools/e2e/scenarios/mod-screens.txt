# Сценарий: создать мир и открыть экраны мода.
#
# Опирается на состояние после tools/e2e/scenarios/boot.txt НЕЛЬЗЯ — каждый
# прогон начинается с главного меню, поэтому шаги идут с самого начала.
#
# Координаты — под окно 1280x720 (run/options.txt) и guiScale 2.

wait_window Minecraft 420000
wait_log 300000 Created: .*atlas/items
windows
wait 8000
shot 00-main-menu

# Singleplayer
move 640 296
click 640 296
wait 4000
shot 01-singleplayer

# В списке миров жмём «Create New World» (его же можно нажать сразу из списка)
move 480 688
click 480 688
wait 3000
shot 02-create-world

# В меню создания мира: «Create New World» внизу.
move 480 688
click 480 688
wait 25000
shot 03-world-loading

jstack in-world
wait_log 120000 Loaded .* world
wait 10000
shot 04-in-world

# Хаб мода — клавиша H (KeybindsRegister.HUB_SCREEN_KEY).
key h
wait 3000
shot 05-hub

# Из хаба открыть вики: клик по иконке вики в панели хаба.
move 640 360
wait 500
shot 06-hub-hover

jstack hub-open
log конец сценария mod-screens
