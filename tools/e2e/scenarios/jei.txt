# Сценарий: настоящий JEI в dev-клиенте.
#
# tools/e2e/run-in-xvfb.sh запускает именно этот сценарий с -PjeiRuntime,
# поэтому в клиенте стоит полный JEI, а не только compile-only API.
#
# Координаты рассчитаны на окно 1280x720 и guiScale 2:
#   * поле поиска JEI — справа внизу (1090, 692);
#   * первый результат JEI — первая ячейка сетки справа (965, 70)
#     по реальному снимку jei-04-jei-overlay.
# После каждого поиска делается снимок результата, а затем снимок страницы U.
# JEI по U открывает первую категорию по сортировке (обычно Information),
# поэтому дополнительно кликаем вкладки категорий 2/3/4 — (503,72), (566,72),
# (614,72) — и снимаем кадры *-tab2 / *-tab3 / *-tab4: там наши категории
# Age of Epoch (например, для души 4-я вкладка — Stat Souls).
# Если раскладка JEI изменится, по этим снимкам координаты легко поправить.

wait_window Minecraft 420000
wait_log 300000 Created: .*atlas/items
wait 8000
shot 00-main-menu

# Главное меню -> одиночная игра -> войти в мир.
#
# В PR workflow JEI идёт третьим сценарием в том же run/: wiki уже создал мир.
# При существующем мире раскладка Select World другая, поэтому сначала выбираем
# первую строку и нажимаем Play Selected World (482,637) — по реальному снимку
# jei-01-world-list. Если JEI запускают отдельно и список пуст, эти клики
# безвредны, а два клика 480,688 ниже создают мир, как в wiki/combat.
move 640 296
click 640 296
wait 4000
shot 01-world-list
move 640 140
click 640 140
wait 500
move 482 637
click 482 637
wait 3000
move 480 688
click 480 688
wait 3000
shot 02-create-world
move 480 688
click 480 688
wait 45000
shot 03-in-world

# Инвентарь: справа должен появиться список предметов JEI.
hold e 500
wait 5000
shot 04-jei-overlay

# (например "Common Helmet Soul"), слова "Stat" в их именах нет, поэтому
# запрос "stat soul" даёт 0 результатов. Ищем душу конкретного слота.
